cmake_minimum_required(VERSION 3.24)
project(suyu_orbis_native LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SO_AOT_CAPTURE_HANDSHAKE "Wait for explicit capture acknowledgments in diagnostic" OFF)
option(SO_SANITIZE "Host ASan/UBSan" OFF)
if(NOT SO_MODULE_BUNDLE)
    message(FATAL_ERROR "SO_MODULE_BUNDLE must reference scripts/emit_native_fixture.py output")
endif()
if(SO_SANITIZE)
    if(SO_PLATFORM_ORBIS)
        message(FATAL_ERROR "Host sanitizers cannot be linked into the Orbis executable")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
set(SO_BUILD_TESTS OFF CACHE BOOL "Run native project's own tests" FORCE)
set(SO_BUILD_UPSTREAM_FIXTURE OFF CACHE BOOL "This project uses a different original fixture" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." infrastructure)
add_library(so_checked_memory STATIC ../src/memory.c)
target_include_directories(so_checked_memory PUBLIC ../include)
add_library(so_aot_machine STATIC ../src/aot.c)
target_link_libraries(so_aot_machine PUBLIC so_checked_memory so_module_bundle)
foreach(target so_checked_memory so_aot_machine)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
if(SO_PLATFORM_ORBIS)
    add_library(so_orbis_video STATIC ../platform/orbis/video_out.cpp)
    target_include_directories(so_orbis_video PUBLIC ../platform/orbis)
    target_link_libraries(so_orbis_video PUBLIC SceVideoOut)
    target_compile_options(so_orbis_video PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(suyu-orbis-aot-render ../experiments/orbis_aot_render/orbis_main.cpp ../tests/aot_runtime_tests.c)
    target_compile_definitions(suyu-orbis-aot-render PRIVATE SO_SELFTEST_NO_MAIN=1)
    if(SO_AOT_CAPTURE_HANDSHAKE)
        target_compile_definitions(suyu-orbis-aot-render PRIVATE SO_AOT_CAPTURE_HANDSHAKE=1)
    endif()
    target_link_libraries(suyu-orbis-aot-render PRIVATE so_aot_machine so_orbis_video)
    target_compile_options(suyu-orbis-aot-render PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(suyu-orbis-aot-render PROPERTIES SUFFIX ".elf")
    find_program(NATIVE_FSELF NAMES create-fself create-fself-macos
        HINTS "${OO_PS4_TOOLCHAIN}/bin/linux" "${OO_PS4_TOOLCHAIN}/bin/macos"
        NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    add_custom_command(TARGET suyu-orbis-aot-render POST_BUILD
        COMMAND "${NATIVE_FSELF}" "-in=$<TARGET_FILE:suyu-orbis-aot-render>"
            "-out=${CMAKE_CURRENT_BINARY_DIR}/suyu-orbis-aot-render.oelf"
            --eboot "${CMAKE_CURRENT_BINARY_DIR}/suyu-orbis-aot-render-eboot.bin"
            --paid 0x3800000000000011
        BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/suyu-orbis-aot-render.oelf"
                   "${CMAKE_CURRENT_BINARY_DIR}/suyu-orbis-aot-render-eboot.bin"
        VERBATIM)
else()
    enable_testing()
    add_executable(memory-tests ../tests/memory_tests.c)
    target_link_libraries(memory-tests PRIVATE so_checked_memory)
    add_test(NAME checked-memory COMMAND memory-tests)
    add_executable(aot-runtime-tests ../tests/aot_runtime_tests.c)
    target_link_libraries(aot-runtime-tests PRIVATE so_aot_machine)
    add_test(NAME real-suyu-aot-runtime COMMAND aot-runtime-tests)
endif()
