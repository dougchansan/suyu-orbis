cmake_minimum_required(VERSION 3.24)
project(suyu_orbis VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SO_SANITIZE "Enable host ASan/UBSan" OFF)
option(SO_BUILD_TESTS "Build host tests" ON)
option(SO_BUILD_UPSTREAM_FIXTURE "Test an imported SYNTHETIC upstream fixture" OFF)
set(SO_MODULE_BUNDLE "" CACHE PATH "Directory written by import_modules.py")
if(SO_PLATFORM_ORBIS)
    set(SO_BUILD_TESTS OFF)
    enable_language(CXX)
endif()
if(SO_SANITIZE)
    if(SO_PLATFORM_ORBIS OR MSVC)
        message(FATAL_ERROR "Sanitizers require a Clang/GCC host build")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(suyu_orbis_registry STATIC src/registry.c)
target_include_directories(suyu_orbis_registry PUBLIC include)
if(MSVC)
    target_compile_options(suyu_orbis_registry PRIVATE /W4 /WX)
else()
    target_compile_options(suyu_orbis_registry PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(so_fixture STATIC tests/fixture.c)
target_include_directories(so_fixture PUBLIC tests)
target_link_libraries(so_fixture PUBLIC suyu_orbis_registry)
add_executable(suyu-orbis-probe src/probe.c)
target_link_libraries(suyu-orbis-probe PRIVATE so_fixture)
if(SO_PLATFORM_ORBIS)
    target_compile_definitions(suyu-orbis-probe PRIVATE SO_PLATFORM_ORBIS=1)
    set_target_properties(suyu-orbis-probe PROPERTIES SUFFIX ".elf")

    # Small graphics gate independent of Suyu: CPU-write two linear display
    # buffers and present them through the real PS4 SceVideoOut ABI.
    add_executable(suyu-orbis-render-probe experiments/orbis_render_probe/render_probe.cpp)
    target_compile_features(suyu-orbis-render-probe PRIVATE cxx_std_17)
    target_compile_options(suyu-orbis-render-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(suyu-orbis-render-probe PRIVATE SceVideoOut)
    set_target_properties(suyu-orbis-render-probe PROPERTIES SUFFIX ".elf")
endif()
if(SO_MODULE_BUNDLE)
    if(NOT EXISTS "${SO_MODULE_BUNDLE}/CMakeLists.txt")
        message(FATAL_ERROR "Run scripts/import_modules.py first")
    endif()
    add_subdirectory("${SO_MODULE_BUNDLE}" generated-bundle)
endif()
if(SO_BUILD_UPSTREAM_FIXTURE)
    if(NOT SO_MODULE_BUNDLE)
        message(FATAL_ERROR "SO_BUILD_UPSTREAM_FIXTURE requires a synthetic module bundle")
    endif()
    add_executable(suyu-orbis-upstream-smoke tests/upstream_smoke.c)
    target_link_libraries(suyu-orbis-upstream-smoke PRIVATE so_module_bundle)
    if(SO_PLATFORM_ORBIS)
        target_compile_definitions(suyu-orbis-upstream-smoke PRIVATE SO_PLATFORM_ORBIS=1)
        set_target_properties(suyu-orbis-upstream-smoke PROPERTIES SUFFIX ".elf")
    elseif(UNIX)
        target_link_libraries(suyu-orbis-upstream-smoke PRIVATE m)
    endif()
endif()
if(SO_BUILD_TESTS)
    enable_testing()
    add_executable(registry-tests tests/registry_tests.c)
    target_link_libraries(registry-tests PRIVATE so_fixture)
    add_test(NAME registry COMMAND registry-tests)
    add_test(NAME probe COMMAND suyu-orbis-probe)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_test(NAME python-tools COMMAND "${Python3_EXECUTABLE}" -m unittest discover
        -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py" -v)
    if(SO_BUILD_UPSTREAM_FIXTURE)
        add_test(NAME upstream-emitted-aarch64 COMMAND suyu-orbis-upstream-smoke)
    endif()
endif()
